Codexion

Working C implementation built on your t_coder / t_program structures. It compiles with -Wall -Wextra -Werror -pthread and was run on fifo and edf with 1 and 5 coders, plus invalid input. Test it harder yourself (helgrind, TSan, valgrind) and make sure you can explain every part, since the defense will ask.

What changed in your structs

r_dongle, l_donglepthread_mutex_t * became t_dongle *
dongle_lockReplaced by write_lock. Each dongle now owns its mutex inside t_dongle
t_program locksPointers became plain values; coders point at them
New in t_coderedf, dongle_cooldown, last_compile_start, start_time
New in t_programThe parsed arguments, monitor thread, dongles array
compilingKept and maintained, but burnout does not need it: last_compile_start is set when the compile starts

Caveats: the code follows the Norm style but run norminette yourself. printf is line-buffered on a terminal but fully buffered into a pipe, and fflush is not in the allowed function list, so for piped output you may want your own write-based logger. Init error paths are minimal.

codexion.h

Your two structs, kept in place. Dongles become t_dongle (mutex + cond + cooldown + heap) so a plain mutex pointer is no longer enough.

main.c

#include "codexion.h"

static void	run_simulation(t_program *p)
{
	int	i;

	i = 0;
	while (i < p->nb_of_coders)
	{
		pthread_create(&p->coders[i].thread, NULL, coder_routine,
			&p->coders[i]);
		i++;
	}
	pthread_create(&p->monitor, NULL, monitor_routine, p);
	pthread_join(p->monitor, NULL);
	i = 0;
	while (i < p->nb_of_coders)
	{
		pthread_join(p->coders[i].thread, NULL);
		i++;
	}
}

int	main(int ac, char **av)
{
	t_program	p;

	memset(&p, 0, sizeof(t_program));
	if (parse_args(ac, av, &p))
	{
		write(2, "Error: invalid arguments\n", 25);
		return (1);
	}
	if (init_program(&p))
	{
		cleanup(&p);
		return (1);
	}
	run_simulation(&p);
	cleanup(&p);
	return (0);
}

parse.c

#include "codexion.h"

static int	parse_num(char *s, size_t *out)
{
	size_t	i;
	size_t	n;

	i = 0;
	n = 0;
	if (!s[0])
		return (1);
	while (s[i])
	{
		if (s[i] < '0' || s[i] > '9')
			return (1);
		n = n * 10 + (s[i] - '0');
		if (n > 2147483647)
			return (1);
		i++;
	}
	*out = n;
	return (0);
}

static int	parse_scheduler(char *s, int *edf)
{
	if (!strcmp(s, "fifo"))
		*edf = 0;
	else if (!strcmp(s, "edf"))
		*edf = 1;
	else
		return (1);
	return (0);
}

int	parse_args(int ac, char **av, t_program *p)
{
	size_t	v[7];
	int		i;

	if (ac != 9)
		return (1);
	i = 0;
	while (i < 7)
	{
		if (parse_num(av[i + 1], &v[i]))
			return (1);
		i++;
	}
	if (v[0] < 1 || v[1] < 1 || parse_scheduler(av[8], &p->edf))
		return (1);
	p->nb_of_coders = v[0];
	p->time_to_burnout = v[1];
	p->time_to_compile = v[2];
	p->time_to_debug = v[3];
	p->time_to_refactor = v[4];
	p->compiles_required = v[5];
	p->dongle_cooldown = v[6];
	return (0);
}

init.c

#include "codexion.h"

static int	init_dongles(t_program *p)
{
	int	i;

	p->dongles = malloc(sizeof(t_dongle) * p->nb_of_coders);
	if (!p->dongles)
		return (1);
	memset(p->dongles, 0, sizeof(t_dongle) * p->nb_of_coders);
	i = 0;
	while (i < p->nb_of_coders)
	{
		pthread_mutex_init(&p->dongles[i].lock, NULL);
		pthread_cond_init(&p->dongles[i].cond, NULL);
		p->dongles[i].queue.data = malloc(sizeof(t_req) * 2);
		if (!p->dongles[i].queue.data)
			return (1);
		i++;
	}
	return (0);
}

static void	init_coder(t_program *p, int i)
{
	t_coder	*c;

	c = &p->coders[i];
	c->id = i + 1;
	c->compiling = 0;
	c->compiles = 0;
	c->nb_of_coders = p->nb_of_coders;
	c->burned_out = &p->burnout_flag;
	c->edf = p->edf;
	c->time_to_burnout = p->time_to_burnout;
	c->time_to_compile = p->time_to_compile;
	c->time_to_debug = p->time_to_debug;
	c->time_to_refactor = p->time_to_refactor;
	c->dongle_cooldown = p->dongle_cooldown;
	c->start_time = p->start_time;
	c->last_compile_start = p->start_time;
	c->l_dongle = &p->dongles[i];
	c->r_dongle = &p->dongles[(i + 1) % p->nb_of_coders];
	c->burnout_lock = &p->burnout_lock;
	c->compile_lock = &p->compile_lock;
	c->write_lock = &p->write_lock;
}

int	init_program(t_program *p)
{
	int	i;

	p->burnout_flag = 0;
	pthread_mutex_init(&p->burnout_lock, NULL);
	pthread_mutex_init(&p->compile_lock, NULL);
	pthread_mutex_init(&p->write_lock, NULL);
	p->start_time = get_time();
	p->coders = malloc(sizeof(t_coder) * p->nb_of_coders);
	if (!p->coders || init_dongles(p))
		return (1);
	i = 0;
	while (i < p->nb_of_coders)
	{
		init_coder(p, i);
		i++;
	}
	return (0);
}

void	cleanup(t_program *p)
{
	int	i;

	i = 0;
	while (p->dongles && i < p->nb_of_coders)
	{
		pthread_mutex_destroy(&p->dongles[i].lock);
		pthread_cond_destroy(&p->dongles[i].cond);
		free(p->dongles[i].queue.data);
		i++;
	}
	free(p->dongles);
	free(p->coders);
	pthread_mutex_destroy(&p->burnout_lock);
	pthread_mutex_destroy(&p->compile_lock);
	pthread_mutex_destroy(&p->write_lock);
}

heap.c

Hand-written binary heap. Order is key, then arrival seq, then coder id, which gives a deterministic tie-break.

#include "codexion.h"

static int	req_less(t_req *a, t_req *b)
{
	if (a->key != b->key)
		return (a->key < b->key);
	if (a->seq != b->seq)
		return (a->seq < b->seq);
	return (a->id < b->id);
}

static void	swap_req(t_req *a, t_req *b)
{
	t_req	tmp;

	tmp = *a;
	*a = *b;
	*b = tmp;
}

void	heap_push(t_heap *h, t_req r)
{
	int	i;

	i = h->size++;
	h->data[i] = r;
	while (i > 0 && req_less(&h->data[i], &h->data[(i - 1) / 2]))
	{
		swap_req(&h->data[i], &h->data[(i - 1) / 2]);
		i = (i - 1) / 2;
	}
}

void	heap_pop(t_heap *h)
{
	int	i;
	int	c;

	h->data[0] = h->data[--h->size];
	i = 0;
	while (2 * i + 1 < h->size)
	{
		c = 2 * i + 1;
		if (c + 1 < h->size && req_less(&h->data[c + 1], &h->data[c]))
			c++;
		if (!req_less(&h->data[c], &h->data[i]))
			break ;
		swap_req(&h->data[c], &h->data[i]);
		i = c;
	}
}

dongle.c

A coder may take a dongle only if it is at the top of the heap, the dongle is free, and the cooldown has passed. FIFO uses key = seq, EDF uses key = deadline.

#include "codexion.h"

static t_req	make_req(t_coder *c, t_dongle *d)
{
	t_req	r;

	r.id = c->id;
	r.seq = d->next_seq++;
	r.key = r.seq;
	if (c->edf)
	{
		pthread_mutex_lock(c->compile_lock);
		r.key = c->last_compile_start + c->time_to_burnout;
		pthread_mutex_unlock(c->compile_lock);
	}
	return (r);
}

static int	can_take(t_coder *c, t_dongle *d)
{
	return (!d->in_use && d->queue.data[0].id == c->id
		&& get_time() >= d->available_at);
}

static void	wait_dongle(t_coder *c, t_dongle *d)
{
	struct timespec	ts;

	if (!d->in_use && d->queue.data[0].id == c->id)
	{
		ts.tv_sec = d->available_at / 1000;
		ts.tv_nsec = (d->available_at % 1000) * 1000000;
		pthread_cond_timedwait(&d->cond, &d->lock, &ts);
	}
	else
		pthread_cond_wait(&d->cond, &d->lock);
}

int	take_dongle(t_coder *c, t_dongle *d)
{
	pthread_mutex_lock(&d->lock);
	heap_push(&d->queue, make_req(c, d));
	while (!is_burned_out(c) && !can_take(c, d))
		wait_dongle(c, d);
	if (is_burned_out(c))
	{
		pthread_mutex_unlock(&d->lock);
		return (1);
	}
	heap_pop(&d->queue);
	d->in_use = 1;
	pthread_mutex_unlock(&d->lock);
	log_state(c, "has taken a dongle");
	return (0);
}

void	release_dongle(t_coder *c, t_dongle *d)
{
	pthread_mutex_lock(&d->lock);
	d->in_use = 0;
	d->available_at = get_time() + c->dongle_cooldown;
	pthread_cond_broadcast(&d->cond);
	pthread_mutex_unlock(&d->lock);
}

coder.c

Dongles are always taken lowest address first, which breaks circular wait. One coder: takes the single dongle, then waits for burnout.

#include "codexion.h"

static int	acquire_both(t_coder *c)
{
	t_dongle	*first;
	t_dongle	*second;

	if (c->l_dongle == c->r_dongle)
	{
		if (take_dongle(c, c->l_dongle))
			return (1);
		while (!is_burned_out(c))
			usleep(500);
		return (1);
	}
	first = c->l_dongle;
	second = c->r_dongle;
	if (first > second)
	{
		first = c->r_dongle;
		second = c->l_dongle;
	}
	if (take_dongle(c, first))
		return (1);
	if (take_dongle(c, second))
		return (release_dongle(c, first), 1);
	return (0);
}

static void	do_compile(t_coder *c)
{
	pthread_mutex_lock(c->compile_lock);
	c->last_compile_start = get_time();
	c->compiling = 1;
	pthread_mutex_unlock(c->compile_lock);
	log_state(c, "is compiling");
	sleep_ms(c, c->time_to_compile);
	pthread_mutex_lock(c->compile_lock);
	c->compiles++;
	c->compiling = 0;
	pthread_mutex_unlock(c->compile_lock);
	release_dongle(c, c->l_dongle);
	release_dongle(c, c->r_dongle);
}

void	*coder_routine(void *arg)
{
	t_coder	*c;

	c = (t_coder *)arg;
	while (!is_burned_out(c))
	{
		if (acquire_both(c))
			break ;
		do_compile(c);
		log_state(c, "is debugging");
		sleep_ms(c, c->time_to_debug);
		log_state(c, "is refactoring");
		sleep_ms(c, c->time_to_refactor);
	}
	return (NULL);
}

monitor.c

Polls every 0.5 ms. Sets the flag and prints under write_lock, then broadcasts every dongle so sleeping threads wake up.

#include "codexion.h"

static int	check_coders(t_program *p)
{
	int		i;
	int		done;
	int		burnt;
	size_t	last;

	i = 0;
	done = 1;
	burnt = 0;
	while (i < p->nb_of_coders)
	{
		pthread_mutex_lock(&p->compile_lock);
		last = p->coders[i].last_compile_start;
		if (p->coders[i].compiles < p->compiles_required)
			done = 0;
		pthread_mutex_unlock(&p->compile_lock);
		if (!burnt && get_time() - last >= p->time_to_burnout)
			burnt = i + 1;
		i++;
	}
	if (done)
		return (-1);
	return (burnt);
}

static void	stop_all(t_program *p, int id)
{
	int	i;

	pthread_mutex_lock(&p->write_lock);
	pthread_mutex_lock(&p->burnout_lock);
	p->burnout_flag = 1;
	pthread_mutex_unlock(&p->burnout_lock);
	if (id > 0)
		printf("%zu %d burned out\n", get_time() - p->start_time, id);
	pthread_mutex_unlock(&p->write_lock);
	i = 0;
	while (i < p->nb_of_coders)
	{
		pthread_mutex_lock(&p->dongles[i].lock);
		pthread_cond_broadcast(&p->dongles[i].cond);
		pthread_mutex_unlock(&p->dongles[i].lock);
		i++;
	}
}

void	*monitor_routine(void *arg)
{
	t_program	*p;
	int			res;

	p = (t_program *)arg;
	while (1)
	{
		res = check_coders(p);
		if (res)
		{
			stop_all(p, res);
			break ;
		}
		usleep(500);
	}
	return (NULL);
}

utils.c

Lock order is dongle lock, then write_lock, then burnout_lock (always innermost).

#include "codexion.h"

size_t	get_time(void)
{
	struct timeval	tv;

	gettimeofday(&tv, NULL);
	return (tv.tv_sec * 1000 + tv.tv_usec / 1000);
}

int	is_burned_out(t_coder *c)
{
	int	flag;

	pthread_mutex_lock(c->burnout_lock);
	flag = *c->burned_out;
	pthread_mutex_unlock(c->burnout_lock);
	return (flag);
}

void	sleep_ms(t_coder *c, size_t ms)
{
	size_t	start;

	start = get_time();
	while (get_time() - start < ms && !is_burned_out(c))
		usleep(500);
}

void	log_state(t_coder *c, char *msg)
{
	pthread_mutex_lock(c->write_lock);
	if (!is_burned_out(c))
		printf("%zu %d %s\n", get_time() - c->start_time, c->id, msg);
	pthread_mutex_unlock(c->write_lock);
}
